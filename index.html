Foil colour variant logic Mediaclip JavaScript

Reads the customer's chosen emboss option, maps it to the correct colour, and applies it as a foil spot colour in the render.

// Read customer's chosen foil colour option
const foil = this.context.getOption("embossColour", "gold");
const embossColours = {
  gold: "#c08b4b",
  silver: "#cacaca",
  holographic: "#e2e2e2",
  green: "#037225",
  blue: "#2330e3",
  red: "#f62e27"
};

collage.addClipart({
  id: "$(package:printondemandworldwide/default)/assets/elements/clipart/shootingstar",
  height: collage.width / 1.5,
  width: collage.width / 1.5,
  x: -320,
  y: -325,
  colorize: embossColours[foil] || embossColours.gold,
  colorizeSpotColor: "FoilStamp"
});

Shop the Collection block WooCommerce PHP / CSS

Reads the current product tags, identifies the matching collection, and renders related products below the product summary without a plugin dependency.

add_action('woocommerce_after_single_product_summary',
           'add_shop_collection_block', 5);

function add_shop_collection_block() {
  if (!is_product()) return;
  global $product;

  $terms = wp_get_post_terms($product->get_id(), 'product_tag');
  $collection = find_collection_slug($terms, 'wedding-collection');

  if (!$collection) return;
  echo do_shortcode('[products limit="8" columns="5" tag="' .
    $collection . '" visibility="any" exclude="' .
    $product->get_id() . '"]');
}
